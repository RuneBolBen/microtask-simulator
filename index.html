function startQualification(){


initials =

document.getElementById("workerInitials").value.trim().toUpperCase();


if(initials.length===0){

alert("Please enter your initials.");

return;

}


document.getElementById("loginScreen").classList.add("hidden");

document.getElementById("qualificationScreen").classList.remove("hidden");


qualIndex = 0;

qualCorrect = 0;


showQualificationQuestion();


}


function showQualificationQuestion(){


const q = qualificationQuestions[qualIndex];


let html =

'<div class="card"><p>'+q.q+'</p>';


q.options.forEach(option => {

html +=

'<button onclick="answerQualification(\''+

option.replace(/'/g,"\\'")+

'\')">'+option+'</button>';

});


html += '</div>';


document.getElementById("qualificationQuestion").innerHTML = html;


}


function answerQualification(answer){


if(answer===qualificationQuestions[qualIndex].answer){

qualCorrect++;

}


qualIndex++;


if(qualIndex < qualificationQuestions.length){

showQualificationQuestion();

return;

}


if(qualCorrect >= 4){


document.getElementById("qualificationScreen").classList.add("hidden");

document.getElementById("workScreen").classList.remove("hidden");


loadTask();


} else {


startLockout();


}


}


function startLockout(){


document.getElementById("qualificationScreen").classList.add("hidden");

document.getElementById("lockoutScreen").classList.remove("hidden");


let remaining = 60;


document.getElementById("lockoutTimer").textContent = remaining;


const lockoutInterval = setInterval(() => {


remaining--;


document.getElementById("lockoutTimer").textContent = remaining;


if(remaining <= 0){


clearInterval(lockoutInterval);


document.getElementById("lockoutScreen").classList.add("hidden");

document.getElementById("qualificationScreen").classList.remove("hidden");


qualIndex = 0;

qualCorrect = 0;


showQualificationQuestion();


}


},1000);


}

}


function submitTask(answer){

"<p class='success'>Task Approved</p>";


}


}else{

document.getElementById("taskFeedback").innerHTML =

"<p class='error'>Incorrect Submission</p>";


}

loadTask();


}
